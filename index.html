# 讀取剛剛生成的 index.html 並將 Banner 改成精準擷取「左半邊 50%」 with open('index.html', 'r', encoding='utf-8') as f: code = f.read() # 使用者明確要求： # 「我只要圖片一半，如同我給你的照片，右半邊空白處不用」 # 原圖是一張超長橫幅圖（1920寬度），右半邊有大片藍天空白或延展部分。 # 使用者只要「左半部」（約 50%～55% 的寬度，剛好是名古屋城、楓樹、以及主要的四個人物）。 # 做法： # 容器維持寬度 100% (手機螢幕寬度，約 390~480px)。 # 把圖片寬度設為 200%（或更精確地讓圖片只顯示左側 50%），並用 overflow: hidden 裁切掉右側 50%！ # 或者使用 CSS 背景圖語法： # background-image: url('31263.png'); # background-size: 200% 100%; (寬度放大到 200%，高度填滿 100%) # background-position: left center; (固定最左邊，因此右邊 50% 完全被排擠到容器外切除！) old_banner_style = """ /* 橫幅容器：使用原圖比例 100% 寬度等比縮放，四角修圓，左右上下完全不裁切任何一人 */ .banner-wrapper { width: 100%; background-color: #79A3D3; border-bottom-left-radius: 26px; border-bottom-right-radius: 26px; overflow: hidden; box-shadow: 0 4px 15px rgba(0, 0, 0, 0.08); line-height: 0; } .banner-img { width: 100%; height: auto; display: block; }""" new_banner_style = """ /* Banner 精準只取左半邊（左半部 50% 滿版），右半邊空白處完全切除 */ .banner-wrapper { width: 100%; height: 220px; background-image: url('31263.png'); background-size: 200% 100%; /* 寬度放大到 200%，高度滿版，精準截取左側 50% */ background-position: left center; /* 靠齊最左側，徹底排除右半邊空白 */ background-repeat: no-repeat; background-color: #79A3D3; border-bottom-left-radius: 28px; border-bottom-right-radius: 28px; overflow: hidden; box-shadow: 0 4px 15px rgba(0, 0, 0, 0.08); position: relative; }""" code = code.replace(old_banner_style, new_banner_style) code = code.replace('', '') # 同時確保 Firebase doc key 更新為全新 key，確保一打開就是最新版 code = code.replace("doc('v2026_full_perfect_v1')", "doc('v2026_half_banner_v1')") with open('index.html', 'w', encoding='utf-8') as f: f.write(code) print("Banner adjusted to left 50% cropped successfully.")
